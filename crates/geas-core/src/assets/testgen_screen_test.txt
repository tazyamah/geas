//! 計画書の「ユースケースの手順」と「画面の具体例」から geas が生成したテスト。手で直さない（計画書を直して作り直す）。
//! 手順は screen_cases.json、実行部は screen_runner.js（Playwright。アプリを知らない汎用のもの）。
//! サーバーは、このクレート自身（geas_server）を新しい DB で起こし、上から順に実行する。node と Playwright が無ければ、何もせず終わる。
//! Chrome 拡張の手順は、拡張が問い合わせてよい先（manifest の host_permissions）のポート 8787 で行う。
use std::net::{Shutdown, TcpListener, TcpStream};
use std::path::PathBuf;
use std::process::{Command, Stdio};
use std::time::Duration;

fn app() -> PathBuf {
    PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("..")
}

/// node と、グローバルに入れた Playwright があれば、NODE_PATH の値を返す
fn node_path() -> Option<String> {
    let root = Command::new("npm").args(["root", "-g"]).output().ok().map(|o| String::from_utf8_lossy(&o.stdout).trim().to_string()).unwrap_or_default();
    let ok = Command::new("node").args(["-e", "require('playwright')"]).env("NODE_PATH", &root).stdout(Stdio::null()).stderr(Stdio::null()).status().map(|s| s.success()).unwrap_or(false);
    if ok {
        Some(root)
    } else {
        None
    }
}

#[test]
fn screen_use_case_steps_pass() {
    let Some(np) = node_path() else {
        eprintln!("node と Playwright が無いので、画面の通しは飛ばします");
        return;
    };
    let ext_ok = TcpListener::bind(("127.0.0.1", 8787)).is_ok();
    let port = if ext_ok { 8787 } else { TcpListener::bind("127.0.0.1:0").unwrap().local_addr().unwrap().port() };
    let work = std::env::temp_dir().join(format!("geas-screen-{}", std::process::id()));
    std::fs::create_dir_all(&work).unwrap();
    // 模擬の返事・時計・開発用の操作パネルを使うので、テスト用の口を開く（--test）
    let mut srv = Command::new(env!("CARGO_BIN_EXE_geas_server"))
        .args(["--port", &port.to_string(), "--db"])
        .arg(work.join("app.sqlite"))
        .arg("--pwa")
        .arg(app().join("ui"))
        .args(["--test", "--dev"])
        .arg(app().join("dev"))
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .spawn()
        .unwrap();
    for _ in 0..100 {
        if let Ok(s) = TcpStream::connect(("127.0.0.1", port)) {
            let _ = s.shutdown(Shutdown::Both);
            break;
        }
        std::thread::sleep(Duration::from_millis(100));
    }
    let here = PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("tests");
    let mut cmd = Command::new("node");
    cmd.arg(here.join("screen_runner.js")).arg(here.join("screen_cases.json")).arg(format!("http://127.0.0.1:{}", port)).arg(&work);
    if !ext_ok {
        cmd.arg("--no-ext");
    }
    let r = cmd.env("NODE_PATH", np).output().unwrap();
    let _ = srv.kill();
    let _ = srv.wait();
    let _ = std::fs::remove_dir_all(&work);
    assert!(r.status.success(), "{}{}", String::from_utf8_lossy(&r.stdout), String::from_utf8_lossy(&r.stderr));
}
